! ============================================================
! CORE-2 — Collapsed Core L3 switch (FINAL, mirror of CORE-1)
! Role: HSRP active for EVEN VLANs (12,14,22,24),
!       HSRP standby for ODD VLANs and shared
! ============================================================

hostname CORE-2
no ip domain-lookup
ip domain name data.com

! ============================================================
